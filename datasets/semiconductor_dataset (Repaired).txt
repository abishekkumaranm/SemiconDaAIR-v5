"""
semiconductor_dataset.py — PyTorch Dataset & Data Pipeline with Multi-Label Degradation Vector.

Supports:
  1. Real Paired KLA Dataset (.npy float32 & images)
  2. Multi-Label Degradation Vector Returns [Speckle, Gaussian, Resolution]
  3. Paired Geometric Augmentations (Identical rotations & flips for Input + GT)
"""

import os
import glob
import numpy as np
import cv2
import torch
from torch.utils.data import Dataset


class SemiconductorDataset(Dataset):
    """
    Paired Semiconductor Inspection Dataset.
    Loads Low-Quality Input [1, H, W] and Ground-Truth Target [1, scale*H, scale*W].
    Optionally returns target degradation labels vector [Speckle, Gaussian, Resolution].
    """
    def __init__(self, gt_dir: str, lq_dir: str, patch_size: int = None, augment: bool = True):
        super().__init__()
        self.gt_dir = gt_dir
        self.lq_dir = lq_dir
        self.patch_size = patch_size
        self.augment = augment

        self.gt_paths = sorted(glob.glob(os.path.join(gt_dir, "*.npy")) + glob.glob(os.path.join(gt_dir, "*.*")))
        self.lq_paths = sorted(glob.glob(os.path.join(lq_dir, "*.npy")) + glob.glob(os.path.join(lq_dir, "*.*")))
        
        self.gt_paths = list(dict.fromkeys(self.gt_paths))
        self.lq_paths = list(dict.fromkeys(self.lq_paths))

        if len(self.gt_paths) == 0 or len(self.lq_paths) == 0:
            raise RuntimeError(f"No valid dataset files found in GT: {gt_dir} or LQ: {lq_dir}")

    def __len__(self):
        return min(len(self.gt_paths), len(self.lq_paths))

    def __getitem__(self, idx):
        gt_path = self.gt_paths[idx]
        lq_path = self.lq_paths[idx]

        if gt_path.endswith(".npy"):
            gt = np.load(gt_path).astype(np.float32)
        else:
            gt = cv2.imread(gt_path, cv2.IMREAD_GRAYSCALE).astype(np.float32) / 255.0

        if lq_path.endswith(".npy"):
            lq = np.load(lq_path).astype(np.float32)
        else:
            lq = cv2.imread(lq_path, cv2.IMREAD_GRAYSCALE).astype(np.float32) / 255.0

        # Paired Augmentations (Identical flip & rot for LQ and GT)
        if self.augment:
            if np.random.rand() < 0.5:
                gt = np.fliplr(gt).copy()
                lq = np.fliplr(lq).copy()
            if np.random.rand() < 0.5:
                gt = np.flipud(gt).copy()
                lq = np.flipud(lq).copy()
            rot_k = np.random.choice([0, 1, 2, 3])
            if rot_k > 0:
                gt = np.rot90(gt, k=rot_k).copy()
                lq = np.rot90(lq).copy()

        tensor_lq = torch.from_numpy(lq).unsqueeze(0)
        tensor_gt = torch.from_numpy(gt).unsqueeze(0)

        # Multi-label degradation ground-truth vector [Speckle, Gaussian, Resolution]
        # By default for 128x128 -> 256x256 paired dataset, all 3 degradations coexist [1, 1, 1]
        deg_vector = torch.tensor([1.0, 1.0, 1.0], dtype=torch.float32)

        return tensor_lq, tensor_gt, deg_vector
